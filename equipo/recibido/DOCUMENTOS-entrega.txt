// Hecho por: DOCUMENTOS
/* Funciones puras para citar documentos en A.L.I.A.S. No es una ruta.
   El texto del documento es DATO: se mete en <datos> y no se obedece.
   Cupo diario: Redis alias:d:<usuario>:<día> con pipe INCR + EXPIRE. */
import { pipe, today } from './_lib.js';

export const MAX_FRAGS = 6;
export const MAX_CHARS = 1500;
export const DOCS_DIA = 20;
const TTL = '172800';

function fallo(clave, status, extra) {
  const e = new Error(clave);
  e.clave = clave;
  e.status = status || 400;
  if (extra) Object.assign(e, extra);
  throw e;
}

/* Quita la marca <datos> para que el documento no pueda cerrar el bloque. */
export function quitarMarca(s) {
  return String(s == null ? '' : s)
    .replace(/<\s*\/?\s*datos\b[^>]*>/gi, ' ')
    .replace(/<\s*\/?\s*datos/gi, ' ');
}

export function sanearTexto(v) {
  let s = String(v == null ? '' : v).slice(0, MAX_CHARS * 4);
  s = s.replace(/[\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f]/g, ' ');
  s = quitarMarca(s).replace(/\s+/g, ' ').trim();
  if (s.length > MAX_CHARS) s = s.slice(0, MAX_CHARS).trim();
  return s;
}

function numPagina(v) {
  const n = Math.floor(Number(v));
  if (!Number.isFinite(n) || n < 1) return 1;
  if (n > 9999) return 9999;
  return n;
}

/* Solo caracteres seguros en la clave. Un usuario raro no arma otra clave. */
export function usuarioClave(usuario) {
  const u = String(usuario == null ? '' : usuario).replace(/[^a-zA-Z0-9_-]/g, '').slice(0, 80);
  if (!u) fallo('srv.docsUsuario', 400);
  return u;
}

export function claveDia(usuario, dia) {
  const d = String(dia || '');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(d)) fallo('srv.docsRoto', 500);
  return 'alias:d:' + usuarioClave(usuario) + ':' + d;
}

/* Máximo 6 fragmentos, cada uno de hasta 1.500 caracteres. No copia campos de más. */
export function validarFragmentos(lista) {
  if (!Array.isArray(lista)) fallo('srv.docsRoto', 400);
  if (lista.length > MAX_FRAGS) fallo('srv.docsMuchos', 400);
  const out = [];
  for (const item of lista) {
    if (!item || typeof item !== 'object' || Array.isArray(item)) fallo('srv.docsRoto', 400);
    const texto = sanearTexto(item.texto != null ? item.texto : item.text);
    if (!texto) continue;
    const pag = item['página'] != null ? item['página'] : (item.pagina != null ? item.pagina : item.page);
    out.push({ pagina: numPagina(pag), texto });
  }
  if (!out.length) fallo('srv.docsVacios', 400);
  return out;
}

/* Bloque para el modelo. Lo que va adentro es DATO, nunca una orden. */
export function bloqueDatos(lista) {
  const frags = validarFragmentos(lista);
  const cuerpo = quitarMarca(frags.map(f => '[p. ' + f.pagina + '] ' + f.texto).join('\n\n'));
  return '<datos>\n' + cuerpo + '\n</datos>';
}

/* Suma 1 al cupo del día. Si se pasa, devuelve el incremento y lanza srv.docsLimite. */
export async function cupoDocs(usuario, max, io) {
  const tope = Number.isFinite(+max) && +max > 0 ? Math.floor(+max) : DOCS_DIA;
  const p = io && typeof io.pipe === 'function' ? io.pipe : pipe;
  const dia = io && typeof io.today === 'function' ? io.today() : today();
  const clave = claveDia(usuario, dia);
  const r = await p([['INCR', clave], ['EXPIRE', clave, TTL]]);
  const n = +r[0] || 0;
  if (n > tope) {
    try { await p([['DECR', clave]]); } catch (e) { /* el tope ya se negó; el contador puede quedar un paso arriba si Redis falla aquí */ }
    fallo('srv.docsLimite', 429, { limit: true });
  }
  return { ok: true, clave, n, dia, max: tope };
}

SUPUESTOS
- El tope de páginas es 40 y el cupo diario es 20. `cupoDocs` acepta otro máximo.
- `k` se recorta a 4..6. Si hay menos aciertos, devuelve menos. Un TXT o CSV sin `\f` es la página 1; `\f` abre otra.
- El `EXPIRE` dura 172800 segundos, igual que los otros cupos diarios de A.L.I.A.S., no la ventana corta de `limit()`.
